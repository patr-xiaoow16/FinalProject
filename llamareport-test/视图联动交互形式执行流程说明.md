## 概述

### 当前状况分析

**现有交互方式**:
- 用户点击选择卡片
- 输入探索问题（可选）
- 点击"生成综合分析"按钮

**存在的局限**:
1. 交互方式单一，仅支持点击选择
2. 用户需要明确知道要分析什么，缺少引导
3. 无法精确定位到图表中的特定区域或数据点
4. 对比分析需要手动思考对比维度
5. 时间序列探索不够直观

**创新目标**:
- 提供多种自然的交互方式
- 降低用户的学习成本和操作门槛
- 支持更精确、更灵活的数据探索
- 智能识别用户意图并推荐下一步操作

## 现有交互方式（已实现）

### 交互形式1: 卡片选择与探索问题

#### 前端实现位置
- 文件: `/frontend/components/VisualizationPanel.vue`
- 关键函数:
  - `handleCardClick()` - 处理卡片点击事件
  - `handleGenerateAnalysis()` - 生成联动分析

#### 后端API端点
- 端点: `POST /api/view-linkage/multi-card`
- 模型: `MultiCardLinkageRequest`
- 处理函数: `multi_card_linkage_analysis()`

#### 执行流程
```
1. 用户点击卡片 → 卡片状态变为selected
2. 用户可选输入探索问题（explorationQuestion）
3. 点击"生成综合分析"按钮
4. 前端构建MultiCardLinkageRequest
5. 调用后端API
6. 后端执行联动分析（对齐分析、综合推理、数据验证、视图生成）
7. 返回ViewLinkageResponse（包含新视图列表、洞察）
8. 前端渲染联动生成的视图卡片
```

#### 数据流

## 新增交互形式详解

### 交互形式2: 图表区域圈选交互

#### 实现概要

**前端实现要点**:
1. 监听Plotly图表的选择事件
2. 提取圈选区域的数据信息
3. 构建包含圈选上下文的联动请求
4. 在原图表上保留圈选标记

**后端处理要点**:
1. 解析圈选区域信息（时间范围、数值范围、数据点）
2. LLM分析圈选区域的特征（上升/下降/异常）
3. 生成针对该区域的联动视图
4. 返回结果时标注圈选区域的洞察

```javascript
// 提取时间范围（如果X轴是时间）
function extractTimeRange(xRange) {
  if (!xRange) return null;

  const [start, end] = xRange;

  // 识别时间格式
  if (typeof start === 'string' && /^\d{4}/.test(start)) {
    // 年份格式：2020, 2021, ...
    return {
      start: start.toString(),
      end: end.toString(),
      granularity: "year"
    };
  } else if (typeof start === 'string' && /Q\d/.test(start)) {
    // 季度格式：2020Q1, 2020Q2, ...
    return {
      start: start,
      end: end,
      granularity: "quarter"
    };
  }

  return null;
}
```

```python
# 在 /backend/agents/view_linkage.py 中添加

async def handle_area_selection_linkage(
    request: ViewLinkageRequest
) -> ViewLinkageResponse:
    """
    处理图表区域圈选联动请求
    """
    # 提取圈选上下文
    click_context = request.click_context
    selected_region = click_context.get("selected_region", {})

    # 分析圈选区域特征
    region_analysis = analyze_selected_region(
        selected_region,
        request.source_view_summary
    )

    # LLM分析圈选意图和特征
    linkage_analysis_prompt = f"""
    用户在图表上圈选了一个区域，需要分析该区域的特征并生成相关视图。

    **圈选区域信息**:
    - 时间范围: {selected_region.get('time_range')}
    - 数值范围: {selected_region.get('value_range')}
    - 数据点数量: {len(selected_region.get('points', []))}
    - 图表类型: {request.source_view_summary.chart_type}
    - 关键指标: {request.source_view_summary.key_metrics}

    **区域特征分析**:
    {region_analysis}

    **任务**:
    1. 识别用户圈选该区域的意图（想验证、想解释、想查看详情）
    2. 分析该区域的数据特征（上升、下降、异常、拐点）
    3. 生成3个联动视图推荐：
       - 验证型视图：验证该区域的数据构成或驱动因素
       - 解释型视图：解释该区域出现的原因或变化机制
       - 导航型视图：导航到相关的文本说明或详细数据

    请以JSON格式返回分析结果。
    """

    # 调用LLM
    llm_response = await Settings.llm.acomplete(linkage_analysis_prompt)
    analysis_result = parse_llm_json_response(llm_response.text)

    # 基于分析结果生成联动视图
    new_views = []

    # 生成验证型视图
    if analysis_result.get("verify_view"):
        verify_view = await generate_verify_view(
            analysis_result["verify_view"],
            request,
            selected_region
        )
        if verify_view:
            new_views.append(verify_view)

    # 生成解释型视图
    if analysis_result.get("explain_view"):
        explain_view = await generate_explain_view(
            analysis_result["explain_view"],
            request,
            selected_region
        )
        if explain_view:
            new_views.append(explain_view)

    # 生成导航型视图
    if analysis_result.get("navigate_view"):
        navigate_view = await generate_navigate_view(
            analysis_result["navigate_view"],
            request,
            selected_region
        )
        if navigate_view:
            new_views.append(navigate_view)

    # 构建响应
    response = ViewLinkageResponse(
        source_view_id=request.source_view_id,
        synthesis_insight={
            "conclusion": analysis_result.get("conclusion", ""),
            "evidence_chain": [],
            "confidence": "high",
            "region_features": region_analysis
        },
        new_views=new_views
    )

    return response


def analyze_selected_region(
    selected_region: Dict[str, Any],
    view_summary: ChartSummary
) -> str:
    """
    分析圈选区域的数据特征
    """
    points = selected_region.get("points", [])
    if not points:
        return "无法分析空区域"

    # 提取Y值（数据值）
    y_values = [p["y"] for p in points]

    # 计算统计特征
    mean_value = sum(y_values) / len(y_values)
    min_value = min(y_values)
    max_value = max(y_values)

    # 识别趋势
    if len(y_values) >= 2:
        # 简单线性趋势
        first_half_avg = sum(y_values[:len(y_values)//2]) / (len(y_values)//2)
        second_half_avg = sum(y_values[len(y_values)//2:]) / (len(y_values) - len(y_values)//2)

        if second_half_avg > first_half_avg * 1.1:
            trend = "上升趋势"
            trend_magnitude = f"+{((second_half_avg / first_half_avg - 1) * 100):.1f}%"
        elif second_half_avg < first_half_avg * 0.9:
            trend = "下降趋势"
            trend_magnitude = f"-{((1 - second_half_avg / first_half_avg) * 100):.1f}%"
        else:
            trend = "平稳"
            trend_magnitude = "波动小于10%"
    else:
        trend = "数据点不足"
        trend_magnitude = "N/A"

    # 识别异常点
    anomalies = []
    threshold = mean_value * 0.2  # 20%偏差视为异常
    for i, point in enumerate(points):
        if abs(point["y"] - mean_value) > threshold:
            anomalies.append(f"{point['x']}: {point['y']}")

    # 构建分析文本
    analysis = f"""
区域特征分析：
- 数据点数量: {len(points)}
- 时间范围: {selected_region.get('time_range', {}).get('start')} 至 {selected_region.get('time_range', {}).get('end')}
- 数值范围: {min_value:.2f} ~ {max_value:.2f}
- 平均值: {mean_value:.2f}
- 趋势: {trend} ({trend_magnitude})
- 异常点: {', '.join(anomalies) if anomalies else '无'}
    """

    return analysis.strip()
```

```javascript
  plotlyDiv.on('plotly_click', function(eventData) {
    if (!eventData || !eventData.points || eventData.points.length === 0) {
      return;

// 推断钻取类型
function inferDrillType(point, chartConfig) {
  const xValue = point.x;
  const yValue = point.y;

  // 判断X轴是否为时间
  const isTimeSeries = /^\d{4}/.test(String(xValue)) || /Q\d/.test(String(xValue));

  // 判断是否为异常点（Y值远高于或低于平均值）
  const avgY = calculateAverageY(chartConfig);
  const isAnomaly = Math.abs(yValue - avgY) > avgY * 0.3;  // 偏差超过30%

  // 判断图表类型
  const chartType = chartConfig.chart_type;

  if (chartType === 'PIE') {
    return 'category_drill';  // 饼图点击 → 分类下钻
  } else if (isTimeSeries) {
    return 'time_drill';  // 时间序列点击 → 时间下钻
  } else if (isAnomaly) {
    return 'anomaly_analysis';  // 异常点点击 → 异常分析
  } else {
    return 'general_drill';  // 通用下钻
  }
}
```

```python
    click_context = request.click_context
    clicked_point = click_context.get("clicked_point", {})
    drill_type = click_context.get("drill_type", "general_drill")

    # 根据钻取类型生成不同的联动视图
    new_views = []

    if drill_type == "time_drill":
        # 时间下钻：生成更细粒度的时间分析
        new_views = await generate_time_drill_views(clicked_point, request)

    elif drill_type == "category_drill":
        # 分类下钻：生成该分类的详细拆分
        new_views = await generate_category_drill_views(clicked_point, request)

    elif drill_type == "anomaly_analysis":
        # 异常分析：解释异常点的原因
        new_views = await generate_anomaly_analysis_views(clicked_point, request)

    else:
        # 通用下钻：生成验证型和解释型视图
        new_views = await generate_general_drill_views(clicked_point, request)

    # 构建响应
    response = ViewLinkageResponse(
        source_view_id=request.source_view_id,
        synthesis_insight={
            "conclusion": f"已生成 {len(new_views)} 个下钻分析视图",
            "evidence_chain": [],
            "confidence": "high"
        },
        new_views=new_views
    )

    return response


async def generate_time_drill_views(
    clicked_point: Dict[str, Any],
    request: ViewLinkageRequest
) -> List[NewViewInfo]:
    """
    生成时间下钻视图
    例如：点击"2024年" → 生成"2024年按季度拆分"
    """
    time_value = clicked_point.get("x")  # 例如 "2024"
    key_metrics = request.source_view_summary.key_metrics

    # LLM生成下钻分析
    drill_prompt = f"""
    用户点击了时间点"{time_value}"，需要生成时间下钻分析视图。

    **当前视图信息**:
    - 图表类型: {request.source_view_summary.chart_type}
    - 关键指标: {key_metrics}
    - 时间粒度: 年度

    **下钻任务**:
    生成3个时间下钻视图：
    1. "{time_value}年按季度拆分" - 将年度数据拆分为Q1, Q2, Q3, Q4
    2. "{time_value}年关键指标对比" - 对比该年度的多个关键指标
    3. "{time_value}年同比变化分析" - 对比{time_value}年与{int(time_value)-1}年的变化

    请为每个视图生成：
    - 视图标题
    - 数据查询（需要从RAG检索什么数据）
    - 图表类型推荐
    - 视图类型（verify/explain/navigate）

    返回JSON格式。
    """

    llm_response = await Settings.llm.acomplete(drill_prompt)
    drill_views = parse_llm_json_response(llm_response.text)

    # 为每个视图检索数据并生成
    new_views = []
    for view_spec in drill_views.get("views", []):
        # 从RAG检索数据
        retrieved_data = await retrieve_data_for_view(view_spec, request)

        # 生成可视化
        if retrieved_data:
            viz_response = await generate_visualization_from_data(
                retrieved_data,
                view_spec
            )

            new_view = NewViewInfo(
                view_id=f"drill_{generate_uuid()}",
                view_type=view_spec.get("view_type", "verify"),
                visualization_response=viz_response,
                description=view_spec.get("title", "时间下钻视图"),
                data_validation={
                    "data_available": True,
                    "data_quality": "high",
                    "data_sources": retrieved_data.get("sources", [])
                },
                related_cards=[request.source_view_id]
            )

            new_views.append(new_view)

    return new_views
```
